private Sub BusquedaBinaria_Click()
    Dim n, i As Integer
    Dim arr() As Integer
    Dim valorBuscado As Integer
    Dim resultado As Integer
    
    ' Leer tamaño del arreglo ingresado por el usuario
    n = Range("B1").Value
    ReDim arr(n - 1)
    
    ' Llenar el arreglo con números aleatorios y ordenarlos
    For i = 0 To n - 1
        arr(i) = Int(100 * Rnd()) + 1
    Next i
    
    ' Ordenar el arreglo usando el método de inserción para asegurarse que esté ordenado
    Call InsertionSort(arr)
    
    ' Mostrar los elementos del arreglo ordenado antes de la búsqueda
    For i = 0 To n - 1
        ActiveSheet.Cells(i + 3, 3).Value = arr(i)
    Next i
    
    ' Solicitar al usuario el valor a buscar
    valorBuscado = InputBox("Ingrese el valor a buscar:")
    
    ' Búsqueda binaria
    resultado = BinarySearch(arr, 0, n - 1, valorBuscado)
    
    ' Mostrar resultados
    If resultado <> -1 Then
        MsgBox "Valor encontrado en la posición: " & resultado + 1
        ActiveSheet.Cells(resultado + 3, 4).Value = "Valor encontrado"
    Else
        MsgBox "Valor no encontrado"
    End If
End Sub

Sub InsertionSort(arr() As Integer)
    Dim n, i, j, key As Integer
    n = UBound(arr) + 1
    For i = 1 To n - 1
        key = arr(i)
        j = i - 1
        While j >= 0 And arr(j) > key
            arr(j + 1) = arr(j)
            j = j - 1
        Wend
        arr(j + 1) = key
    Next i
End Sub

Function BinarySearch(arr() As Integer, l As Integer, r As Integer, x As Integer) As Integer
    Dim mid As Integer
    While l <= r
        mid = l + (r - l) \ 2
        If arr(mid) = x Then
            BinarySearch = mid
            Exit Function
        ElseIf arr(mid) < x Then
            l = mid + 1
        Else
            r = mid - 1
        End If
    Wend
    BinarySearch = -1
End Function

Private Sub MergeSort_click(arr() As Integer, l As Integer, r As Integer)
    If l < r Then
        ' Encuentra el punto medio del arreglo
        Dim m As Integer
        m = l + (r - l) \ 2
        
        ' Ordena la primera y segunda mitad recursivamente
        MergeSort arr, l, m
        MergeSort arr, m + 1, r
        
        ' Une las mitades ordenadas
        Merge arr, l, m, r
    End If
End Sub

Sub Merge(arr() As Integer, l As Integer, m As Integer, r As Integer)
    Dim n1 As Integer, n2 As Integer
    n1 = m - l + 1
    n2 = r - m
    
    ' Crea arreglos temporales
    Dim leftArr() As Integer, rightArr() As Integer
    ReDim leftArr(n1 - 1)
    ReDim rightArr(n2 - 1)
    
    ' Copia datos a los arreglos temporales
    Dim i As Integer, j As Integer
    For i = 0 To n1 - 1
        leftArr(i) = arr(l + i)
    Next i
    For j = 0 To n2 - 1
        rightArr(j) = arr(m + 1 + j)
    Next j
    
    ' Combina los arreglos temporales en el arreglo original
    i = 0
    j = 0
    Dim k As Integer
    k = l
    While i < n1 And j < n2
        If leftArr(i) <= rightArr(j) Then
            arr(k) = leftArr(i)
            i = i + 1
        Else
            arr(k) = rightArr(j)
            j = j + 1
        End If
        k = k + 1
    Wend
    
    ' Copia los elementos restantes de leftArr
    While i < n1
        arr(k) = leftArr(i)
        i = i + 1
        k = k + 1
    Wend
    
    ' Copia los elementos restantes de rightArr
    While j < n2
        arr(k) = rightArr(j)
        j = j + 1
        k = k + 1
    Wend
End Sub

Private Sub QuickSort_clik(arr() As Integer, low As Integer, high As Integer)
    ' Verifica si el índice inferior es menor que el índice superior
    If low < high Then
        ' Encuentra el índice de partición
        Dim pi As Integer
        pi = Partition(arr, low, high)

        ' Ordena recursivamente los elementos antes y después de la partición
        QuickSort arr, low, pi - 1
        QuickSort arr, pi + 1, high
    End If
End Sub

Function Partition(arr() As Integer, low As Integer, high As Integer) As Integer
    Dim pivot As Integer
    pivot = arr(high) ' Toma el último elemento como pivote
    Dim i As Integer
    i = low - 1 ' Índice de los elementos menores que el pivote
    
    ' Recorre el arreglo desde el índice bajo hasta el alto menos uno
    For j = low To high - 1
        ' Si el elemento actual es menor o igual al pivote
        If arr(j) <= pivot Then
            i = i + 1 ' Incrementa el índice de los elementos menores
            ' Intercambia arr(i) con arr(j)
            Dim temp As Integer
            temp = arr(i)
            arr(i) = arr(j)
            arr(j) = temp
        End If
    Next j
    
    ' Intercambia el pivote con el elemento en arr(i + 1)
    Dim temp As Integer
    temp = arr(i + 1)
    arr(i + 1) = arr(high)
    arr(high) = temp
    
    ' Retorna el índice de partición
    Partition = i + 1
End Function
